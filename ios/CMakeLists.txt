cmake_minimum_required(VERSION 3.20)
project(th15_ios LANGUAGES C CXX OBJC OBJCXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
 message(FATAL_ERROR "Configure with ios/build_ios.sh")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
get_filename_component(TH15_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(TH15_DEPENDENCIES "${TH15_ROOT}/.local/dependencies" CACHE PATH "SDL source")
set(TH15_ASSETS "${TH15_ROOT}/.local/assets" CACHE PATH "Private resources")
set(TH15_BUILD_NUMBER "7" CACHE STRING "Build number")
set(TH15_BUNDLE_ID "org.th15.native" CACHE STRING "Bundle ID")
set(TH15_ICON_CATALOG "${TH15_ROOT}/.local/app-icon/Assets.xcassets" CACHE PATH "Private original icon")
set(TH15_SDL_LIBRARY "" CACHE FILEPATH "Optional previously verified SDL3 static library")
option(TH15_DIAGNOSTICS "Scripted simulator input" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_METAL OFF CACHE BOOL "" FORCE)
set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_OPENGLES ON CACHE BOOL "" FORCE)
set(SDL_HIDAPI OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
if(TH15_SDL_LIBRARY)
 add_library(SDL3::SDL3 STATIC IMPORTED)
 set_target_properties(SDL3::SDL3 PROPERTIES IMPORTED_LOCATION "${TH15_SDL_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${TH15_DEPENDENCIES}/SDL/include")
else()
 add_subdirectory("${TH15_DEPENDENCIES}/SDL" "${CMAKE_BINARY_DIR}/SDL")
endif()
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${TH15_ROOT}/th15_web/cpp/game/*.cpp")
file(GLOB HOST_SOURCES CONFIGURE_DEPENDS "${TH15_ROOT}/th15_web/cpp/sdl/*.cpp")
list(FILTER HOST_SOURCES EXCLUDE REGEX "/BrowserRuntime\.cpp$")
add_executable(th15 MACOSX_BUNDLE main.mm NativeHost.cpp Diagnostics.mm ${GAME_SOURCES} ${HOST_SOURCES} "${TH15_ROOT}/portable/sdl/Renderer.cpp")
if(EXISTS "${TH15_ICON_CATALOG}/AppIcon.appiconset/Contents.json")
 target_sources(th15 PRIVATE "${TH15_ICON_CATALOG}")
 set_source_files_properties("${TH15_ICON_CATALOG}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
 set_target_properties(th15 PROPERTIES XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME "AppIcon")
endif()
target_compile_definitions(th15 PRIVATE TH_NATIVE_PLATFORM=1 TH15_IOS=1 TH15_DEVELOPMENT_HARNESS=0 TH15_DIAGNOSTICS=$<BOOL:${TH15_DIAGNOSTICS}>)
target_compile_options(th15 PRIVATE -fno-strict-aliasing -fno-fast-math -ffp-contract=off -fsigned-char -Wno-unknown-attributes -Wno-deprecated-declarations $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
set_source_files_properties(main.mm Diagnostics.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
target_link_libraries(th15 PRIVATE SDL3::SDL3 "-framework UIKit" "-framework Foundation" "-framework AVFoundation" "-framework QuartzCore" "-framework OpenGLES")
if(TH15_SDL_LIBRARY)
 target_link_libraries(th15 PRIVATE "-framework CoreMedia" "-framework CoreVideo" "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreGraphics" "-framework CoreMotion" "-framework GameController" "-framework Metal" "-weak_framework CoreHaptics")
endif()
target_link_options(th15 PRIVATE "LINKER:-dead_strip")
configure_file(Info.plist.in "${CMAKE_BINARY_DIR}/TH15-Info.plist" @ONLY)
set_target_properties(th15 PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_BINARY_DIR}/TH15-Info.plist" XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${TH15_BUNDLE_ID}" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2" XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "14.0" XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "NO" XCODE_ATTRIBUTE_ENABLE_BITCODE "NO")
if(NOT EXISTS "${TH15_ASSETS}/th15.dat")
 message(FATAL_ERROR "Prepare original TH15 resources first")
endif()
add_custom_command(TARGET th15 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${TH15_ASSETS}" "$<TARGET_BUNDLE_DIR:th15>/assets" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "$<TARGET_BUNDLE_DIR:th15>/licenses")
